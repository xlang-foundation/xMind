cmake_minimum_required(VERSION 3.24)
project(AgentFlowNative LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(AGENTFLOW_XLANG3_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../xlang3" CACHE PATH "Sibling xlang3 source checkout")
add_library(agentflow_secrets STATIC src/secret_protection.cpp)
target_include_directories(agentflow_secrets PUBLIC include)
if(WIN32)
  target_link_libraries(agentflow_secrets PRIVATE Crypt32)
endif()
set(AGENTFLOW_XLANG3_RUNTIME_DIR "${AGENTFLOW_XLANG3_SOURCE}/build/Release" CACHE PATH "Existing Release runtime directory")
add_library(agentflow_xlang3 SHARED IMPORTED GLOBAL)
if(WIN32)
  set(AGENTFLOW_XLANG3_LIBRARY "${AGENTFLOW_XLANG3_RUNTIME_DIR}/xlang3_runtime.lib" CACHE FILEPATH "xlang3 import library")
  set(AGENTFLOW_XLANG3_SHARED "${AGENTFLOW_XLANG3_RUNTIME_DIR}/xlang3_runtime.dll" CACHE FILEPATH "xlang3 runtime DLL")
  set_target_properties(agentflow_xlang3 PROPERTIES IMPORTED_IMPLIB "${AGENTFLOW_XLANG3_LIBRARY}")
  target_compile_definitions(agentflow_xlang3 INTERFACE XLANG3_USE_SHARED)
else()
  set(AGENTFLOW_XLANG3_SHARED "${AGENTFLOW_XLANG3_RUNTIME_DIR}/libxlang3_runtime${CMAKE_SHARED_LIBRARY_SUFFIX}" CACHE FILEPATH "xlang3 shared runtime")
endif()
if(NOT EXISTS "${AGENTFLOW_XLANG3_SHARED}")
  message(FATAL_ERROR "Build/select the existing xlang3 Release runtime before configuring AgentFlow")
endif()
set_target_properties(agentflow_xlang3 PROPERTIES IMPORTED_LOCATION "${AGENTFLOW_XLANG3_SHARED}")
target_include_directories(agentflow_xlang3 INTERFACE "${AGENTFLOW_XLANG3_SOURCE}/sdk")
add_library(agentflow_core STATIC src/backend_lease.cpp src/xlang_sqlite.cpp src/repository.cpp)
target_include_directories(agentflow_core PUBLIC include)
target_link_libraries(agentflow_core PRIVATE agentflow_xlang3 agentflow_secrets)
if(MSVC)
  target_compile_options(agentflow_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(agentflow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
# Production persistence must call SQLite through embedded xlang3.
add_executable(xmind_persistence_demo demos/persistence_demo.cpp)
target_link_libraries(xmind_persistence_demo PRIVATE agentflow_core)
if(WIN32)
  add_custom_command(TARGET xmind_persistence_demo POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "${AGENTFLOW_XLANG3_SHARED}" "$<TARGET_FILE_DIR:xmind_persistence_demo>")
endif()
if(BUILD_TESTING)
  if(WIN32)
    add_executable(agentflow_secret_contract tests/secret_protection_contract.cpp)
    target_link_libraries(agentflow_secret_contract PRIVATE agentflow_secrets)
    add_test(NAME native_secret_protection_contract COMMAND agentflow_secret_contract)
    set_tests_properties(native_secret_protection_contract PROPERTIES TIMEOUT 30)
  endif()
  if(WIN32)
    set(default_python_source "C:/Python/Python314/Lib")
  else()
    set(default_python_source "")
  endif()
  set(AGENTFLOW_PYTHON_LIB_SOURCE "${default_python_source}" CACHE PATH "Standard-library source for xlang3; no interpreter")
  if(NOT EXISTS "${AGENTFLOW_PYTHON_LIB_SOURCE}/sqlite3/__init__.py")
    message(FATAL_ERROR "Set AGENTFLOW_PYTHON_LIB_SOURCE to the allowed standard-library source directory")
  endif()
  add_executable(agentflow_xlang_sqlite_contract tests/xlang_sqlite_contract.cpp)
  target_link_libraries(agentflow_xlang_sqlite_contract PRIVATE agentflow_core)
  if(WIN32)
    add_custom_command(TARGET agentflow_xlang_sqlite_contract POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${AGENTFLOW_XLANG3_SHARED}" "$<TARGET_FILE_DIR:agentflow_xlang_sqlite_contract>")
  endif()
  add_test(NAME embedded_xlang_sqlite_contract COMMAND agentflow_xlang_sqlite_contract
    "${AGENTFLOW_XLANG3_RUNTIME_DIR}/modules" "${AGENTFLOW_PYTHON_LIB_SOURCE}")
  set_tests_properties(embedded_xlang_sqlite_contract PROPERTIES TIMEOUT 30)
  add_executable(agentflow_repository_contract tests/repository_contract.cpp)
  target_link_libraries(agentflow_repository_contract PRIVATE agentflow_core)
  if(WIN32)
    add_custom_command(TARGET agentflow_repository_contract POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${AGENTFLOW_XLANG3_SHARED}" "$<TARGET_FILE_DIR:agentflow_repository_contract>")
  endif()
  add_test(NAME xlang_repository_contract COMMAND agentflow_repository_contract
    "${AGENTFLOW_XLANG3_RUNTIME_DIR}/modules" "${AGENTFLOW_PYTHON_LIB_SOURCE}")
  set_tests_properties(xlang_repository_contract PROPERTIES TIMEOUT 60)
  if(WIN32)
    add_executable(agentflow_credential_repository_contract tests/credential_repository_contract.cpp)
    target_link_libraries(agentflow_credential_repository_contract PRIVATE agentflow_core)
    add_custom_command(TARGET agentflow_credential_repository_contract POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "${AGENTFLOW_XLANG3_SHARED}" "$<TARGET_FILE_DIR:agentflow_credential_repository_contract>")
    add_test(NAME encrypted_credential_repository_contract COMMAND agentflow_credential_repository_contract
      "${AGENTFLOW_XLANG3_RUNTIME_DIR}/modules" "${AGENTFLOW_PYTHON_LIB_SOURCE}")
    set_tests_properties(encrypted_credential_repository_contract PROPERTIES TIMEOUT 60)
  endif()
endif()
option(AGENTFLOW_BUILD_DIRECT_SQLITE_REFERENCE "Build superseded direct-SQLite reference" OFF)
if(AGENTFLOW_BUILD_DIRECT_SQLITE_REFERENCE)
set(sqlite_dir "${AGENTFLOW_XLANG3_SOURCE}/third_party/sqlite")
# Reuse only the public-domain SQLite amalgamation, not xlang3's Python binding.
file(SHA256 "${sqlite_dir}/sqlite3.c" sqlite_c_hash)
file(SHA256 "${sqlite_dir}/sqlite3.h" sqlite_h_hash)
if(NOT sqlite_c_hash STREQUAL "5057c9ac94775dba2d4619e2ce151962e2c0a137d5cee9c77d42d242bd3e004e" OR
   NOT sqlite_h_hash STREQUAL "3c63651aee18741e2d0cf97a8bce891905929d31121c8b2a42ba15084903ae85")
  message(FATAL_ERROR "SQLite baseline changed; audit and explicitly update the native dependency pin")
endif()
add_library(agentflow_sqlite STATIC "${sqlite_dir}/sqlite3.c")
target_include_directories(agentflow_sqlite PUBLIC "${sqlite_dir}")
target_compile_definitions(agentflow_sqlite PRIVATE SQLITE_THREADSAFE=1)
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(agentflow_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
endif()
add_library(agentflow_store_direct_reference STATIC src/store.cpp)
target_include_directories(agentflow_store_direct_reference PUBLIC include)
target_link_libraries(agentflow_store_direct_reference PUBLIC agentflow_core PRIVATE agentflow_sqlite)
if(MSVC)
  target_compile_options(agentflow_store_direct_reference PRIVATE /W4 /EHsc)
else()
  target_compile_options(agentflow_store_direct_reference PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(agentflow_store_contract tests/store_contract.cpp)
  target_link_libraries(agentflow_store_contract PRIVATE agentflow_store_direct_reference agentflow_sqlite)
  add_test(NAME native_store_contract COMMAND agentflow_store_contract)
  set_tests_properties(native_store_contract PROPERTIES TIMEOUT 90)
endif()
endif()
