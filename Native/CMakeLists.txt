if(WIN32)
  # Runtime-directory changes must not silently reuse cached files from a
  # different SDK generation. Identical relocated artifacts remain supported.
  foreach(sdk_file SHARED LIBRARY)
    if(sdk_file STREQUAL "SHARED")
      set(sdk_expected "${AGENTFLOW_XLANG3_RUNTIME_DIR}/xlang3_runtime.dll")
    else()
      set(sdk_expected "${AGENTFLOW_XLANG3_RUNTIME_DIR}/xlang3_runtime.lib")
    endif()
    if(NOT EXISTS "${sdk_expected}" OR NOT EXISTS "${AGENTFLOW_XLANG3_${sdk_file}}")
      message(FATAL_ERROR "The selected SDK runtime artifacts are incomplete")
    endif()
    file(SHA256 "${sdk_expected}" sdk_expected_sha)
    file(SHA256 "${AGENTFLOW_XLANG3_${sdk_file}}" sdk_actual_sha)
    if(NOT sdk_expected_sha STREQUAL sdk_actual_sha)
      message(FATAL_ERROR "Cached SDK ${sdk_file} differs from AGENTFLOW_XLANG3_RUNTIME_DIR; clear or update the cached DLL/import-library paths")
    endif()
  endforeach()
endif()
