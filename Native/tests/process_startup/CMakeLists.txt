cmake_minimum_required(VERSION 3.25)
project(xmind_process_startup_diagnostic LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This diagnostic exercises the native Windows process adapter")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(XMIND_NATIVE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(xmind_process_startup_diagnostic process_startup_diagnostic.cpp "${XMIND_NATIVE_ROOT}/src/windows_process.cpp")
target_include_directories(xmind_process_startup_diagnostic PRIVATE "${XMIND_NATIVE_ROOT}/include" "${XMIND_NATIVE_ROOT}/third_party")
target_link_libraries(xmind_process_startup_diagnostic PRIVATE bcrypt)
if(MSVC)
  target_compile_options(xmind_process_startup_diagnostic PRIVATE /utf-8 /EHsc)
endif()
